//! Capabilities with a legitimate bootstrap authority (WO Â§3.A):
//! - Space::new_root() seeds Pid(0) as the ROOT principal holding GRANT over
//!   the root Path("") resource. Pid(0) is kernel-only; it never runs as a task.
//! - Every grant flows from a holder of GRANT on a covering resource.
//! - Attenuation strictly shrinks; empty subsets rejected; handles unique and
//!   never recycled into confusion; revoked/dead handles -> DenyStale forever.
//! Directory-boundary path covering (Batch 2) retained verbatim.

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Pid(pub u64);
#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
pub struct IdentityId(pub u64);
#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Kind { Kernel, User, Service, App, Agent }
#[derive(Debug, Clone, Copy)]
pub struct Identity { pub id: IdentityId, pub kind: Kind, pub owner: Option<IdentityId> }
#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct CapHandle(pub u64);
#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Resource {
    Path { prefix: [u8; 64], len: usize },
    Endpoint { id: u64 },
    Device { id: u64 },
    Memory { region: u64 },
}
#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Decision { Allow, DenyNoCap, DenyStale, DenyInsufficient }

pub const R_READ: u8 = 1;
pub const R_WRITE: u8 = 2;
pub const R_GRANT: u8 = 4;
pub const R_CONNECT: u8 = 8;
pub const R_OPEN: u8 = 16;
/// A right never grantable to user principals; reserved for the kernel root.
pub const R_ROOT: u8 = 32;

pub const ROOT_PID: Pid = Pid(0);

#[derive(Clone, Copy)]
struct Cap {
    handle: CapHandle,
    owner: Pid,
    resource: Resource,
    rights: u8,
    live: bool,
}

pub struct Space {
    caps: [Option<Cap>; 128],
    next_handle: u64,
    next_identity: u64,
}

fn path_res(s: &str) -> Resource {
    let mut p = [0u8; 64];
    let l = s.len().min(64);
    p[..l].copy_from_slice(&s.as_bytes()[..l]);
    Resource::Path { prefix: p, len: l }
}

impl Space {
    /// Bootstrap model: Pid(0) holds ROOT|GRANT|RW over the root path ("").
    /// This is the ONLY seed of authority in the system; it is kernel-held.
    pub fn new_root() -> Self {
        let mut sp = Space { caps: [None; 128], next_handle: 1, next_identity: 1 };
        let root = path_res("");
        let _ = sp.insert_raw(ROOT_PID, root, R_READ | R_WRITE | R_GRANT | R_ROOT);
        sp
    }

    pub fn new_identity(&mut self, kind: Kind, owner: Option<IdentityId>) -> Identity {
        let id = IdentityId(self.next_identity);
        self.next_identity += 1;
        Identity { id, kind, owner }
    }

    pub fn check(&self, pid: Pid, res: &Resource, rights: u8) -> Decision {
        let mut dead_cover = false;
        let mut live_partial = false;
        for c in self.caps.iter().flatten() {
            if c.owner != pid || !resource_covers(c.resource, *res) {
                continue;
            }
            if !c.live {
                dead_cover = true;
                continue;
            }
            if c.rights & rights == rights {
                return Decision::Allow;
            }
            live_partial = true;
        }
        if live_partial { Decision::DenyInsufficient }
        else if dead_cover { Decision::DenyStale }
        else { Decision::DenyNoCap }
    }

    /// Grant requires the granter to hold (rights | R_GRANT) on a covering
    /// resource. R_ROOT itself can never be granted onward: it is filtered
    /// here so no principal below the kernel can mint root authority.
    pub fn grant(&mut self, from: Pid, to: Pid, res: Resource, rights: u8) -> Result<CapHandle, Decision> {
        if rights & R_ROOT != 0 {
            return Err(Decision::DenyInsufficient);
        }
        if rights == 0 {
            return Err(Decision::DenyInsufficient);
        }
        let need = rights | R_GRANT;
        match self.check(from, &res, need) {
            Decision::Allow => self.insert_raw(to, res, rights),
            other => Err(other),
        }
    }

    pub fn attenuate(&mut self, pid: Pid, h: CapHandle, subset: u8) -> Result<CapHandle, Decision> {
        let slot = self.slot_of(pid, h).ok_or(Decision::DenyStale)?;
        let (resource, rights) = {
            let c = self.caps[slot].expect("slot_of checked");
            (c.resource, c.rights)
        };
        if subset == 0 || subset & !rights != 0 {
            return Err(Decision::DenyInsufficient);
        }
        self.insert_raw(pid, resource, subset)
    }

    pub fn revoke(&mut self, pid: Pid, h: CapHandle) -> Result<(), Decision> {
        let slot = self.slot_of(pid, h).ok_or(Decision::DenyStale)?;
        if let Some(c) = self.caps[slot].as_mut() {
            c.live = false;
        }
        Ok(())
    }

    pub fn revoke_all(&mut self, pid: Pid) -> usize {
        let mut n = 0;
        if pid == ROOT_PID {
            return 0; // kernel root authority is not revocable by process exit
        }
        for c in self.caps.iter_mut().flatten() {
            if c.owner == pid && c.live {
                c.live = false;
                n += 1;
            }
        }
        n
    }

    pub fn live_count(&self) -> u64 {
        self.caps.iter().flatten().filter(|c| c.live).count() as u64
    }

    fn slot_of(&self, pid: Pid, h: CapHandle) -> Option<usize> {
        self.caps.iter().position(|c| matches!(c, Some(c) if c.handle == h && c.owner == pid && c.live))
    }

    fn insert_raw(&mut self, owner: Pid, resource: Resource, rights: u8) -> Result<CapHandle, Decision> {
        let slot = self.caps.iter().position(|c| c.is_none()).ok_or(Decision::DenyNoCap)?;
        let h = CapHandle(self.next_handle);
        self.next_handle += 1;
        self.caps[slot] = Some(Cap { handle: h, owner, resource, rights, live: true });
        Ok(h)
    }
}

fn resource_covers(granted: Resource, requested: Resource) -> bool {
    match (granted, requested) {
        (Resource::Path { prefix: g, len: gl }, Resource::Path { prefix: r, len: rl }) => {
            if rl < gl { return false; }
            if g[..gl] != r[..gl] { return false; }
            if gl == rl { return true; }
            gl == 64 || r[gl] == b'/'
        }
        (Resource::Endpoint { id: a }, Resource::Endpoint { id: b }) => a == b,
        (Resource::Device { id: a }, Resource::Device { id: b }) => a == b,
        (Resource::Memory { region: a }, Resource::Memory { region: b }) => a == b,
        _ => false,
    }
}

//! WO Â§3.B corrections on top of Batch 2:
//! - No duplicate frame ever handed out twice (cursor never rewinds).
//! - alloc_contiguous() for task stacks: verifies N physically adjacent
//!   frames in ONE pass and LEAKS NOTHING on failure (all frames back).
//! - All arithmetic checked; alignment enforced; region bounds validated.
//! - free() is now REAL: frames are handed back via a small hole stack and are
//!   re-issued (LIFO) before fresh cursor frames â€” no double-issue because a
//!   returned frame leaves the hole stack only once.

use crate::memory::Region;

pub const PAGE: u64 = 4096;

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum FrameError { NoMemory }

pub struct FrameAlloc {
    regions: [Extent; 32],
    region_count: usize,
    cursor_region: usize,
    cursor_frame: u64,
    free: u64,
    holes: [u64; 128],   // returned frames, re-issued LIFO
    hole_count: usize,
}

#[derive(Clone, Copy)]
struct Extent { start: u64, end: u64 }

impl FrameAlloc {
    pub fn new(map: &[Region], reserved_end: u64, ceiling: u64) -> Result<Self, FrameError> {
        let mut regions = [Extent { start: 0, end: 0 }; 32];
        let mut region_count = 0usize;
        let mut free = 0u64;
        for r in map.iter() {
            if r.kind != 1 { continue; }
            let start = r.start.max(reserved_end).next_multiple_of(PAGE);
            let end = r.end.min(ceiling) & !(PAGE - 1);
            if start < end && region_count < regions.len() {
                regions[region_count] = Extent { start, end };
                region_count += 1;
                free = free.checked_add((end - start) / PAGE).ok_or(FrameError::NoMemory)?;
            }
        }
        if region_count == 0 { return Err(FrameError::NoMemory); }
        Ok(Self {
            regions, region_count, cursor_region: 0,
            cursor_frame: regions[0].start,
            free,
            holes: [0; 128], hole_count: 0,
        })
    }

    /// Single-frame allocation. Holes (freed frames) are re-issued first.
    /// Invariant: a frame is in exactly one of {holes, cursor-not-yet-passed}
    /// â€” never both, so it can never be issued twice.
    pub fn alloc(&mut self) -> Result<u64, FrameError> {
        if self.hole_count > 0 {
            self.hole_count -= 1;
            let f = self.holes[self.hole_count];
            if self.free > 0 { self.free -= 1; }
            return Ok(f);
        }
        let mut scanned = 0usize;
        loop {
            let ext = self.regions[self.cursor_region];
            if self.cursor_frame + PAGE <= ext.end {
                let frame = self.cursor_frame;
                self.cursor_frame += PAGE;
                if self.free > 0 { self.free -= 1; }
                return Ok(frame);
            }
            self.cursor_region = (self.cursor_region + 1) % self.region_count;
            self.cursor_frame = self.regions[self.cursor_region].start;
            scanned += 1;
            if scanned > self.region_count { return Err(FrameError::NoMemory); }
        }
    }

    /// Contiguous N-frame allocation (WO Â§3.B). One pass; on ANY failure all
    /// frames acquired in this call are returned via free() â€” zero leaks.
    pub fn alloc_contiguous(&mut self, n: usize) -> Result<u64, FrameError> {
        if n == 0 { return Err(FrameError::NoMemory); }
        let mut got: [u64; 32] = [0; 32];
        if n > got.len() { return Err(FrameError::NoMemory); }
        for i in 0..n {
            match self.alloc() {
                Ok(f) => got[i] = f,
                Err(e) => {
                    // return everything we already took: no leak on failure
                    for j in 0..i { let _ = self.free(got[j]); }
                    return Err(e);
                }
            }
        }
        // verify physical adjacency
        for i in 1..n {
            if got[i] != got[0].checked_add((i as u64) * PAGE).ok_or(FrameError::NoMemory)? {
                for j in 0..n { let _ = self.free(got[j]); }
                return Err(FrameError::NoMemory);
            }
        }
        Ok(got[0])
    }

    /// Return a frame. Only accepts frames inside a managed extent; frames
    /// not from this allocator are rejected (Err) rather than corrupting state.
    pub fn free(&mut self, f: u64) -> Result<(), FrameError> {
        if f % PAGE != 0 { return Err(FrameError::NoMemory); }
        let ok = self.regions[..self.region_count].iter().any(|e| f >= e.start && f < e.end);
        if !ok { return Err(FrameError::NoMemory); }
        if self.hole_count >= self.holes.len() { return Err(FrameError::NoMemory); }
        self.holes[self.hole_count] = f;
        self.hole_count += 1;
        self.free = self.free.checked_add(PAGE).ok_or(FrameError::NoMemory)?;
        Ok(())
    }

    pub fn free_bytes(&self) -> u64 { self.free }
    pub fn total_bytes(&self) -> u64 {
        self.regions[..self.region_count].iter().map(|e| e.end - e.start).sum()
    }
}

/// Allocate a kernel stack via alloc_contiguous (fails with NO leak). The
/// frame below stack_base is never allocated => touching it faults (guard).
pub fn alloc_stack() -> Result<(u64, u64, [u64; STACK_PAGES]), TaskError> {
    let st = crate::kernel_state();
    let base = st.frames.alloc_contiguous(STACK_PAGES).map_err(|_| TaskError::NoStack)?;
    let mut frames = [0u64; STACK_PAGES];
    for i in 0..STACK_PAGES {
        frames[i] = base + (i as u64) * 4096;

    Ok((base + (STACK_PAGES as u64) * 4096, guard, frames))

/// Spawn a kernel task. Frame layout (single source of truth):
///   [stack_top-16] = entry      <- initial rsp; switcher ret lands here
///   [stack_top-8 ] = task_exit <- entry's ret target
/// SAFETY: stack written exclusively before the task is scheduled.
pub fn spawn_kernel(name: &'static str, entry: Entry) -> Result<usize, TaskError> {
    let (stack_top, _guard, frames) = alloc_stack()?;
    let tid;
    // SAFETY: boot-ordered table write; failure path frees nothing because
    // the table slot is found BEFORE the stack frames are committed.
    unsafe {
        let slot = TASKS.iter_mut().position(|t| t.state == State::Free).ok_or(TaskError::TooMany)?;
        slot.tid = NEXT_TID;
        tid = slot.tid;
        NEXT_TID += 1;
        slot.name = name;
        slot.entry = Some(entry);
        slot.stack_base = stack_top - (STACK_PAGES as u64) * 4096;
        slot.stack_top = stack_top;
        slot.state = State::Ready;
        slot.ticks_run = 0;
        slot.quantums = 0;
        slot.stack_frames = frames;
        let top = stack_top as *mut u64;
        core::ptr::write_volatile(top.offset(-2), entry as usize as u64);    // top-16
        core::ptr::write_volatile(top.offset(-1), task_exit as usize as u64); // top-8
        slot.saved_rsp = stack_top - 16;
    }
    Ok(tid)
}

        }
    }

/// SAFETY: (1) never called from an IRQ handler â€” enforced by call graph:

/// task stacks; (3) single core. Full register state is preserved because

/// spilled by the compiler before any yield point in generated code. The only

/// a cooperative switcher. (A preemptive switcher must save the full frame;
/// that is exactly why preemption is NOT claimed here.)
pub fn switch_to(next: usize) {
    let current = current_idx();
    if current == next { return; }
    // SAFETY: see contract above.
    unsafe {
        TASKS[current].state = State::Ready;
        TASKS[next].state = State::Running;
        TASKS[next].quantums += 1;
        TASKS[current].ticks_run = 0;
        CURRENT = next;
        PER_CPU.current_tid = TASKS[next].tid as u32;
        let cur = core::ptr::read_volatile(core::ptr::addr_of!(SWITCH_COUNT));
        core::ptr::write_volatile(core::ptr::addr_of!(SWITCH_COUNT), cur + 1);
        let saved = core::ptr::addr_of_mut!(TASKS[current].saved_rsp) as *mut u64;
        let target = core::ptr::read_volatile(core::ptr::addr_of!(TASKS[next].saved_rsp));
        core::arch::asm!(
            "mov %rsp, (%rax)\n",
            "mov %rbx, %rsp\n",
            "ret\n",
            in("rax") saved,
            in("rbx") target,
        );
    }
}

/// slot, switch away. SAFETY: runs on the dying task's stack; frames are
/// returned BEFORE the switch (the allocator never touches task stacks).

    // SAFETY: reclaim frames then slot.
    unsafe {
        for f in TASKS[current].stack_frames.iter() {
            if *f != 0 {
                let _ = st.frames.free(*f);
            }

        let slot = core::ptr::addr_of_mut!(TASKS[current]);

        switch_to(next);
    } else {
        crate::fail("no runnable task after exit");
    }
    unreachable!()
}

/// Syscall body: EUK int-0x80 ABI (documented, versioned):
///   rax = number, rdi = a0, rsi = a1, rdx = a2; result -> rax.
///   0 SYS_ABI_VERSION -> ABI_VERSION, 1 SYS_GETPID, 2 SYS_GETCAPS,
///   3 SYS_IPC_CREATE, 4 SYS_IPC_SEND, 5 SYS_GETTICKS.

/// thunk (iretq restores the full CPU frame) â€” WO Â§4.8.

    let pid = crate::process::current_pid();

        Ok(v) => v,
        Err(e) => crate::syscall::errno_code(e) as u64 | 0x8000_0000_0000_0000,
    }
}

//!  1. TSS descriptor fully written AND validated BEFORE ltr (deviation D2).
//!  2. Correct GDT TSS descriptor (base/limit/type=0x89 available-64bit-TSS).

/// sgdt into a 10-byte scratch and read the base. WO Â§4 fix for gdt_addr().
/// SAFETY: sgdt writes {u16 limit, u64 base}; the scratch is on our stack.
fn gdt_addr() -> u64 {
    let mut scratch = [0u8; 10];
    unsafe {
        asm!("sgdt [{}]", in(reg) scratch.as_mut_ptr(), options(nostack));
    }
    u64::from_le_bytes([
        scratch[2], scratch[3], scratch[4], scratch[5],
        scratch[6], scratch[7], scratch[8], scratch[9],
    ])
}

fn tss_addr() -> u64 { core::ptr::addr_of!(euk_tss) as u64 }

/// Write + validate the TSS GDT descriptor, fill TSS rsp0, then ltr.
/// Validation BEFORE ltr (WO Â§4.1): every field is re-read and decoded before
/// the load. SAFETY: kernel-only structures; called once with ints disabled.
unsafe fn setup_tss_and_ltr() -> bool {
    unsafe {
        let gdt = gdt_addr();
        let tss = tss_addr();
        let limit: u32 = 103; // sizeof 64-bit TSS - 1
        let d = (gdt + (GDT_TSS_SELECTOR as u64 & !7)) as *mut u8;
        let b = tss.to_le_bytes();
        // 16-byte TSS descriptor (GDT slots 5/6):
        core::ptr::write_volatile(d.offset(0), (limit & 0xff) as u8);
        core::ptr::write_volatile(d.offset(1), ((limit >> 8) & 0xff) as u8);
        core::ptr::write_volatile(d.offset(2), b[0]);
        core::ptr::write_volatile(d.offset(3), b[1]);
        core::ptr::write_volatile(d.offset(4), b[2]);
        core::ptr::write_volatile(d.offset(5), 0x89); // present | 64-bit TSS available
        core::ptr::write_volatile(d.offset(6), ((limit >> 16) & 0x0f) as u8);
        core::ptr::write_volatile(d.offset(7), b[3]);
        core::ptr::write_volatile(d.offset(8), b[4]);
        core::ptr::write_volatile(d.offset(9), b[5]);
        core::ptr::write_volatile(d.offset(10), b[6]);
        core::ptr::write_volatile(d.offset(11), b[7]);
        // TSS rsp0: kernel stack used on ring-3 entries
        let tssp = tss as *mut u64;
        core::ptr::write_volatile(tssp.offset(1), core::ptr::addr_of!(ring3_kstack_top) as u64);
        // VALIDATE before ltr
        let v0 = core::ptr::read_volatile(d.offset(0)) as u64;
        let v1 = core::ptr::read_volatile(d.offset(1)) as u64;
        let vlim = (v0 & 0xff) | ((v1 & 0xff) << 8);
        if vlim != (limit & 0xffff) { return false; }
        if core::ptr::read_volatile(d.offset(5)) != 0x89 { return false; }
        let lo = (core::ptr::read_volatile(d.offset(2)) as u64)
            | ((core::ptr::read_volatile(d.offset(3)) as u64) << 8)
            | ((core::ptr::read_volatile(d.offset(4)) as u64) << 16)
            | ((core::ptr::read_volatile(d.offset(7)) as u64) << 24);
        let hi = (core::ptr::read_volatile(d.offset(8)) as u64)
            | ((core::ptr::read_volatile(d.offset(9)) as u64) << 8)
            | ((core::ptr::read_volatile(d.offset(10)) as u64) << 16)
            | ((core::ptr::read_volatile(d.offset(11)) as u64) << 24);
        if lo | (hi << 32) != tss { return false; }
        asm!("ltr %ax", in("ax") GDT_TSS_SELECTOR, options(nostack));
        true
    }
}

pub fn enter(map: &vmm::UserMapping) -> bool {

        asm!("cli", options(nomem, nostack));

/// Observer: reads the data frame (physical identity access by the kernel) and
/// reports whether the round trip completed. data layout: [0]=ticks,
/// [8]=ABI-ok marker (1), [16]=sequence marker (2).
pub fn observe(data_frame: u64, deadline_ticks: u64) -> (bool, bool, bool) {
    let p = data_frame as *const u64;
    let mut abi_ok = false;
    let mut ticks_ok = false;
    let mut done = false;
    while crate::intr::ticks() < deadline_ticks {
        // SAFETY: kernel reading its own allocated frame.
        unsafe {
            let m1 = core::ptr::read_volatile(p.offset(1)) as u32;
            let m2 = core::ptr::read_volatile(p.offset(2)) as u32;
            let t = core::ptr::read_volatile(p.offset(0));
            if m1 == 1 { abi_ok = true; }
            if t > 0 { ticks_ok = true; }
            if m2 == 2 { done = true; }
        }
        if done {
            break;
        }
        crate::task::check_preempt();
    }
    (abi_ok, ticks_ok, done)
}

//! Kernel selftests, now 18, all gating BOOT_OK (exit 35 on any FAIL under
//! --test-exit). Tests 1-17 as Batch 2 (with root-authority semantics), plus
//! test 18: ring-3 round trip read from the observer-published result.

use crate::{serial, serial_num};
use crate::capability::{Decision, Pid, Resource, R_READ, R_WRITE, ROOT_PID, R_CONNECT};

fn path(s: &str) -> Resource {
    let mut p = [0u8; 64];
    p[..s.len()].copy_from_slice(s.as_bytes());
    Resource::Path { prefix: p, len: s.len() }
}

fn check(n: u8, name: &str, ok: bool) {
    serial("[TEST ");
    serial_num(n as u64);
    serial("] ");
    if ok {
        serial("PASS ");
        serial(name);
        serial("\n");
    } else {
        serial("FAIL ");
        serial(name);
        serial("\n");
    }
}

extern "C" {
    // kernel.rs publishes the observer verdict via this static
    static RING3_RESULT: (bool, bool, bool);
}

pub fn run_all() -> bool {
    let st = crate::kernel_state();
    let mut all = true;

    let f1 = st.frames.alloc().is_ok();
    all &= f1;
    check(1, "frame-alloc", f1);

    let heap_ok = unsafe { crate::heap::alloc(&mut st.heap, 64).is_ok() };
    all &= heap_ok;
    check(2, "heap-alloc", heap_ok);

    let t0 = crate::intr::ticks();
    let mut advanced = false;
    for _ in 0..1_000_000u32 {
        if crate::intr::ticks() != t0 {
            advanced = true;
            break;
        }
    }
    all &= advanced;
    check(3, "timer-advance", advanced);

    let switched = crate::task::has_switched();
    all &= switched;
    check(4, "sched-switch", switched);

    // 5-9: capability battery with ROOT authority semantics
    let proc_a = Pid(2);
    let g = st.caps.grant(ROOT_PID, proc_a, path("/data"), R_READ);
    all &= g.is_ok();
    check(5, "cap-grant", g.is_ok());
    if let Ok(h) = g {
        let allowed = matches!(st.caps.check(proc_a, &path("/data/f"), R_READ), Decision::Allow);
        all &= allowed;
        check(6, "cap-allow", allowed);
        let denied = matches!(st.caps.check(proc_a, &path("/data/f"), R_WRITE), Decision::DenyInsufficient);
        all &= denied;
        check(7, "cap-write-deny", denied);
        let att = st.caps.attenuate(proc_a, h, R_READ);
        let att_ok = att.is_ok()
            && matches!(st.caps.check(proc_a, &path("/data/f"), R_READ), Decision::Allow)
            && matches!(st.caps.check(proc_a, &path("/data/f"), R_WRITE), Decision::DenyInsufficient);
        all &= att_ok;
        check(8, "cap-attenuate", att_ok);
        let revoked = st.caps.revoke(proc_a, h).is_ok()
            && matches!(st.caps.check(proc_a, &path("/data/f"), R_READ), Decision::DenyStale);
        all &= revoked;
        check(9, "cap-revoke-stale", revoked);
    } else {
        check(6, "cap-allow", false);
        check(7, "cap-write-deny", false);
        check(8, "cap-attenuate", false);
        check(9, "cap-revoke-stale", false);
        all = false;
    }

    // 10-14: IPC battery (root grants Connect on the endpoint to the owner)
    let ep = crate::ipc::create_endpoint(Pid(1)).ok();
    all &= ep.is_some();
    check(10, "ipc-create", ep.is_some());
    if let Some(epid) = ep {
        let ep_res = Resource::Endpoint { id: epid };
        let cg = st.caps.grant(ROOT_PID, Pid(1), ep_res, R_CONNECT);
        let sent = cg.is_ok() && crate::ipc::send(Pid(1), epid, b"ping", None).is_ok();
        all &= sent;
        check(11, "ipc-send-auth", sent);
        let stranger = crate::ipc::send(Pid(42), epid, b"x", None);
        let denied_ipc = matches!(stranger, Err(crate::ipc::IpcError::NoAccess));
        all &= denied_ipc;
        check(12, "ipc-send-deny", denied_ipc);
        let closed = crate::ipc::close_all_owned(Pid(1));
        all &= closed == 1;
        check(13, "ipc-cleanup", closed == 1);
        let peer_dead = matches!(crate::ipc::send(Pid(1), epid, b"y", None), Err(crate::ipc::IpcError::PeerDead));
        all &= peer_dead;
        check(14, "ipc-peer-dead", peer_dead);
    } else {
        check(11, "ipc-send-auth", false);
        check(12, "ipc-send-deny", false);
        check(13, "ipc-cleanup", false);
        check(14, "ipc-peer-dead", false);
        all = false;
    }

    // 15: stale handle rejection after owner death
    let stale = {
        match st.caps.grant(ROOT_PID, Pid(7), path("/tmp"), R_READ) {
            Ok(h) => {
                st.caps.revoke_all(Pid(7));
                st.caps.revoke(Pid(7), h).is_err()
            }
            Err(_) => false,
        }
    };
    all &= stale;
    check(15, "stale-handle-reject", stale);

    // 16: fs open is capability-checked (deny by default for unprivileged)
    let fs_deny = matches!(crate::fs::open(Pid(2), "/etc/motd", false), Err(crate::fs::FsError::Deny));
    all &= fs_deny;
    check(16, "fs-open-deny", fs_deny);

    // 17: mount state honest - only true after a REAL fat32 mount
    if crate::fs::is_mounted() {
        check(17, "fs-mounted", true);
    } else {
        serial("[TEST 17] SKIP fs-mounted (no device)\n");
    }

    // 18: ring-3 + syscall round trip (observer-published verdict)

        core::ptr::read_volatile(core::ptr::addr_of!(RING3_RESULT))
    };
    let r3 = abi_ok && ticks_ok && done;
    all &= r3;
    check(18, "ring3-syscall-roundtrip", r3);

    all
}
