cmake_minimum_required(VERSION 3.16)
project(ExileLivingBattlefield LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
set(CMAKE_CXX_FLAGS_RELEASE "-O3")
find_package(PNG REQUIRED)
find_package(Threads REQUIRED)
add_executable(living_battlefield
  src/main.cpp src/config.cpp src/daylight.cpp src/atlas.cpp src/png_io.cpp src/scene.cpp src/renderer.cpp)
target_include_directories(living_battlefield PRIVATE src)
target_compile_options(living_battlefield PRIVATE -Wall -Wextra -Wno-unused-parameter)
target_link_libraries(living_battlefield PRIVATE PNG::PNG Threads::Threads)
